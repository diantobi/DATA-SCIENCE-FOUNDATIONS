####VARIABLES AND DATA TYPES

age = 22

myage = "22"

#### LISTS
shopping = ["Milk", "Bread", "Eggs", "Rice"]
print(shopping)

fruits = ["Apple", "Orange", "Banana"]
print(fruits)
print(fruits[0])
print(fruits[1])
fruits[1] = "Mango"
print(fruits)
fruits.append("Pineapple")
print(fruits)
print(len(fruits))
print(fruits[-1])
