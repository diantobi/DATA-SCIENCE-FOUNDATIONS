###Financial Example: Stock Returns
prices = [100, 102, 101, 104, 103, 107, 109, 108, 111, 115]

df = pd.DataFrame({
    "Price": prices
})

print(df)
df["Return"] = df["Price"].pct_change()
print(df)

plt.plot(df["Price"])
plt.title("Stock Price")
plt.xlabel("Trading Day")
plt.ylabel("Price")
plt.show()

plt.plot(df["Return"])
plt.title("Daily Stock Returns")
plt.xlabel("Trading Day")
plt.ylabel("Return")
plt.show()

plt.hist(df["Return"].dropna(), bins=5)
plt.title("Distribution of Daily Returns")
plt.xlabel("Daily Return")
plt.ylabel("Frequency")
plt.show()
